//go:build openlog_node_fixtures

// Generates test/interop/go-sampler-fixtures.json for the Node.js agent from the Go agent's
// sampler (package openlog, unexported API). It is not part of agents/go: scripts/gen-go-fixtures.sh
// copies this file into a temporary copy of agents/go and runs
//
//	OPENLOG_NODE_FIXTURES=<out.json> go test -tags openlog_node_fixtures -run TestGenerateNodeFixtures .
package openlog

import (
	"context"
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"os"
	"testing"

	"go.opentelemetry.io/otel/attribute"
	"go.opentelemetry.io/otel/propagation"
	sdktrace "go.opentelemetry.io/otel/sdk/trace"
	"go.opentelemetry.io/otel/trace"
)

type nodeFixtureSpan struct {
	Sampled       bool     `json:"sampled"`
	Traceparent   string   `json:"traceparent"`
	Tracestate    string   `json:"tracestate"`
	SamplingRatio *float64 `json:"samplingRatio"`
}

type nodeFixture struct {
	Name     string  `json:"name"`
	Ratio    float64 `json:"ratio"`
	WriteRV  bool    `json:"writeRV"`
	TraceID  string  `json:"traceId"`
	SpanID   string  `json:"spanId"`
	ChildID  string  `json:"childSpanId"`
	Random   string  `json:"randomness,omitempty"` // value returned by the rv generator (14 hex)
	Incoming *struct {
		Traceparent string `json:"traceparent"`
		Tracestate  string `json:"tracestate"`
	} `json:"incoming"`
	Span  nodeFixtureSpan `json:"span"`
	Child nodeFixtureSpan `json:"child"`
}

type fixtureIDs struct {
	traceID        trace.TraceID
	spanID, second trace.SpanID
	n              int
}

func (g *fixtureIDs) NewIDs(context.Context) (trace.TraceID, trace.SpanID) {
	return g.traceID, g.next()
}
func (g *fixtureIDs) NewSpanID(context.Context, trace.TraceID) trace.SpanID { return g.next() }
func (g *fixtureIDs) next() trace.SpanID {
	g.n++
	if g.n == 1 {
		return g.spanID
	}
	return g.second
}

func fixtureSpan(ctx context.Context, span trace.Span) nodeFixtureSpan {
	c := propagation.MapCarrier{}
	propagation.TraceContext{}.Inject(ctx, c)
	out := nodeFixtureSpan{Sampled: span.SpanContext().IsSampled(), Traceparent: c.Get("traceparent"), Tracestate: c.Get("tracestate")}
	s, ok := span.(sdktrace.ReadOnlySpan) // dropped spans are non-recording and carry no attributes
	if !ok {
		return out
	}
	for _, kv := range s.Attributes() {
		if kv.Key == SamplingRatioKey && kv.Value.Type() == attribute.FLOAT64 {
			v := kv.Value.AsFloat64()
			out.SamplingRatio = &v
		}
	}
	return out
}

func TestGenerateNodeFixtures(t *testing.T) {
	out := os.Getenv("OPENLOG_NODE_FIXTURES")
	if out == "" {
		t.Skip("OPENLOG_NODE_FIXTURES not set")
	}
	tid := func(r uint64) string {
		var id [16]byte
		copy(id[:8], []byte{0x4b, 0xf9, 0x2f, 0x35, 0x77, 0xb3, 0x4d, 0xa6})
		binary.BigEndian.PutUint64(id[8:], r|0x5a<<56)
		return hex.EncodeToString(id[:])
	}
	type in = struct {
		Traceparent string `json:"traceparent"`
		Tracestate  string `json:"tracestate"`
	}
	remote := func(flags, ts string) *in {
		return &in{Traceparent: "00-0af7651916cd43dd8448eb211c80319c-b7ad6b7169203331-" + flags, Tracestate: ts}
	}
	cases := []nodeFixture{
		{Name: "root ratio 1", Ratio: 1, TraceID: tid(0x12345678)},
		{Name: "root ratio 0", Ratio: 0, TraceID: tid(0xffffffffffffff)},
		{Name: "root ratio 0.25 sampled", Ratio: 0.25, TraceID: tid(0xd0000000000000)},
		{Name: "root ratio 0.25 at threshold", Ratio: 0.25, TraceID: tid(0xc0000000000000)},
		{Name: "root ratio 0.25 dropped", Ratio: 0.25, TraceID: tid(0xbfffffffffffff)},
		{Name: "root ratio 0.1 sampled", Ratio: 0.1, TraceID: tid(0xf0000000000000)},
		{Name: "root ratio 0.001 sampled", Ratio: 0.001, TraceID: tid(0xffffffffffffff)},
		{Name: "root ratio 0.5 writeRV sampled", Ratio: 0.5, WriteRV: true, TraceID: tid(0), Random: "80000000000001"},
		{Name: "root ratio 0.5 writeRV dropped", Ratio: 0.5, WriteRV: true, TraceID: tid(0xffffffffffffff), Random: "00000000000001"},
		{Name: "root ratio 1 writeRV", Ratio: 1, WriteRV: true, TraceID: tid(1), Random: "0123456789abcd"},
		{Name: "remote sampled level1 th", Ratio: 1, TraceID: tid(1), Incoming: remote("01", "ot=th:c")},
		{Name: "remote sampled level2 th rv vendor", Ratio: 0.25, TraceID: tid(1), Incoming: remote("03", "ot=th:8;rv:abcdefabcdefab,congo=t61rcWkgMzE")},
		{Name: "remote sampled legacy p", Ratio: 1, TraceID: tid(1), Incoming: remote("01", "ot=p:2")},
		{Name: "remote sampled no ot", Ratio: 0.5, TraceID: tid(1), Incoming: remote("01", "vendor=value")},
		{Name: "remote sampled th 0", Ratio: 1, TraceID: tid(1), Incoming: remote("03", "ot=th:0")},
		{Name: "remote not sampled", Ratio: 1, TraceID: tid(1), Incoming: remote("02", "ot=rv:00000000000001")},
		{Name: "remote sampled writeRV keeps state", Ratio: 0.5, WriteRV: true, TraceID: tid(1), Random: "11111111111111", Incoming: remote("03", "ot=th:c;x:y")},
	}
	defer func(orig func() (uint64, string)) { newRandomness = orig }(newRandomness)
	for i := range cases {
		c := &cases[i]
		traceID, _ := trace.TraceIDFromHex(c.TraceID)
		ids := &fixtureIDs{traceID: traceID, spanID: trace.SpanID{0x01, 0x02, 0x03, 0x04, 0x05, 0x06, 0x07, 0x08}, second: trace.SpanID{0x0a, 0x0b, 0x0c, 0x0d, 0x0e, 0x0f, 0x10, 0x11}}
		c.SpanID = hex.EncodeToString(ids.spanID[:])
		c.ChildID = hex.EncodeToString(ids.second[:])
		if c.Random != "" {
			rv := c.Random
			r, _ := hex.DecodeString("00" + rv)
			newRandomness = func() (uint64, string) { return binary.BigEndian.Uint64(r), rv }
		}
		sdk := sdktrace.NewTracerProvider(sdktrace.WithSampler(newSampler(c.Ratio, c.WriteRV)), sdktrace.WithIDGenerator(ids))
		tr := randomTracerProvider{tp: sdk}.Tracer("fixtures")
		ctx := context.Background()
		if c.Incoming != nil {
			ctx = propagation.TraceContext{}.Extract(ctx, propagation.MapCarrier{"traceparent": c.Incoming.Traceparent, "tracestate": c.Incoming.Tracestate})
		}
		sctx, span := tr.Start(ctx, "span", trace.WithSpanKind(trace.SpanKindServer))
		cctx, child := tr.Start(sctx, "child", trace.WithSpanKind(trace.SpanKindClient))
		c.Span = fixtureSpan(sctx, span)
		c.Child = fixtureSpan(cctx, child)
		child.End()
		span.End()
	}
	b, err := json.MarshalIndent(map[string]any{
		"generator": "agents/node/scripts/gen-go-fixtures.sh (Go agent sampler.go, package openlog)",
		"cases":     cases,
	}, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
